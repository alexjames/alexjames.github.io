## Graphs
```
class GraphNode:
    def __init__(self, val = 0, neighbors = None):
        self.val = val
        self.neighbors = neighbors if neighbors is not None else []


def dfs(node):
    visited = set()
    _dfs(node, visited)

def _dfs(node, visited):
    if node in visited:
        return
    visited.add(node)
    print("{}: {}".format(node.val, [n.val for n in node.neighbors]))
    for n in node.neighbors:
        _dfs(n, visited)


def build_graph(adjacency_list: dict[str, list[str]]) -> dict[str, GraphNode]:
    # Create one Node object for every unique value.
    nodes = {
        val: GraphNode(val) for val in adjacency_list
    }

    # Connect the nodes.
    for val, neighbors in adjacency_list.items():
        nodes[val].neighbors = [
            nodes[neighbor] for neighbor in neighbors
        ]

    return nodes

# Example usage
adjacency_list = {
    "zebra": ["apple", "cat"],
    "apple": ["zebra", "dog"],
    "cat": ["zebra"],
    "dog": ["apple"],
}

x = build_graph(adjacency_list)

dfs(x.get("zebra"))
```
